/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  /** Configure the main internal regulator output voltage
  */
  __HAL_RCC_PWR_CLK_ENABLE();
  __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE1);

  /** Initialize the RCC oscillators.
   *  25 MHz HSE quartz through the PLL: M=25 (VCO input 1 MHz),
   *  N=432 (VCO 432 MHz), P=2 -> SYSCLK 216 MHz, Q=9 -> 48 MHz spare.
   *  All derived timing constants (I2C TIMINGR, TIM7 prescaler, PWM
   *  prescalers, stepper ARR mapping) are documented in ROBOT_CONFIGS.h.
  */
  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSE;
  RCC_OscInitStruct.HSEState = RCC_HSE_ON;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSE;
  RCC_OscInitStruct.PLL.PLLM = 25U;
  RCC_OscInitStruct.PLL.PLLN = 432U;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV2;
  RCC_OscInitStruct.PLL.PLLQ = 9U;
  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  /* Over-drive is required above 180 MHz (VOS1 + over-drive). */
  if (HAL_PWREx_EnableOverDrive() != HAL_OK)
  {
    Error_Handler();
  }

  /** Initializes the CPU, AHB and APB buses clocks
  */
  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK|RCC_CLOCKTYPE_SYSCLK
                              |RCC_CLOCKTYPE_PCLK1|RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV4;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV2;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_7) != HAL_OK)
  {
    Error_Handler();
  }
}

/**
  * @brief I2C1 Initialization Function
  * @param None
  * @retval None
  */
static void MX_I2C1_Init(void)
{
  hi2c1.Instance = I2C1;
  /* I2C1 kernel clock = PCLK1 (54 MHz); TIMINGR in ROBOT_CONFIGS.h assumes it
     (see BSP_i2c1_init in robot_board.c).
     OwnAddress1 = 0x14 => 7-bit slave address 0x0A (HAL stores it shifted). */
  hi2c1.Init.Timing = ROBOT_I2C_TIMING_100K;
  hi2c1.Init.OwnAddress1 = 0x14U; /* 7-bit 0x0A, matches the RPi driver */
  hi2c1.Init.AddressingMode = I2C_ADDRESSINGMODE_7BIT;
  hi2c1.Init.DualAddressMode = I2C_DUALADDRESS_DISABLE;
  hi2c1.Init.OwnAddress2 = 0;
  hi2c1.Init.OwnAddress2Masks = I2C_OA2_NOMASK;
  hi2c1.Init.GeneralCallMode = I2C_GENERALCALL_DISABLE;
  hi2c1.Init.NoStretchMode = I2C_NOSTRETCH_DISABLE;
  if (HAL_I2C_Init(&hi2c1) != HAL_OK)
  {
    Error_Handler();
  }
  if (HAL_I2CEx_ConfigAnalogFilter(&hi2c1, I2C_ANALOGFILTER_ENABLE) != HAL_OK)
  {
    Error_Handler();
  }
  if (HAL_I2CEx_ConfigDigitalFilter(&hi2c1, 0) != HAL_OK)
  {
    Error_Handler();
  }
}

/** @brief DeInit/Init cycle after a bus error, then re-arms listen mode. */
static void ROBOT_i2c_recover(void)
{
  if (i2c_recover_req == 0U)
  {
    return;
  }
  (void)HAL_I2C_DeInit(&hi2c1);
  if (HAL_I2C_Init(&hi2c1) != HAL_OK)
  {
    return; /* retry on the next loop pass */
  }
  (void)HAL_I2CEx_ConfigAnalogFilter(&hi2c1, I2C_ANALOGFILTER_ENABLE);
  (void)HAL_I2CEx_ConfigDigitalFilter(&hi2c1, 0U);
  i2c_recover_req = 0U;
  ROBOT_i2c_arm_listen();
}

/** @brief One-time application bring-up: steppers, encoders, protocol, motion,
 *         OLED splash, watchdog, I2C listen mode. */
static void ROBOT_init(void)
{
  /* Steppers (ids as wired; left pair {2,3}, right pair {0,1}) ---------- */
  STEPPERS_MJ_Init(0U, &htim12, TIM_CHANNEL_1, EN4_Pin, GPIOB, DIR4_Pin, GPIOB, GPIO_PIN_SET);
  STEPPERS_MJ_Init(1U, &htim4,  TIM_CHANNEL_1, EN3_Pin, GPIOB, DIR3_Pin, GPIOB, GPIO_PIN_SET);
  STEPPERS_MJ_Init(2U, &htim3,  TIM_CHANNEL_1, EN2_Pin, GPIOC, DIR2_Pin, GPIOA, GPIO_PIN_RESET);
  STEPPERS_MJ_Init(3U, &htim2,  TIM_CHANNEL_1, EN1_Pin, GPIOB, DIR1_Pin, GPIOA, GPIO_PIN_RESET);
  STEPPERS_MJ_Init_left_steppers(left_motors_id, 2U, &htim14, left_motors_reach_target_callback);
  STEPPERS_MJ_Init_right_steppers(right_motors_id, 2U, &htim13, right_motors_reach_target_callback);
  STEPPERS_MJ_disable_left_steppers();
  STEPPERS_MJ_disable_right_steppers();

  /* Encoders and the 100 Hz odometry sampling timer --------------------- */
  ENCODERS_MJ_Init(0U, &htim5, 0U);
  ENCODERS_MJ_Init(1U, &htim8, 0U);
  (void)HAL_TIM_Base_Start_IT(&htim7);

  /* Protocol + motion engine -------------------------------------------- */
  MOTION_init();
  PROTO_init();

  /* RGB status LED ------------------------------------------------------- */
  (void)HAL_TIM_PWM_Start(&htim11, TIM_CHANNEL_1);
  (void)HAL_TIM_PWM_Start(&htim9, TIM_CHANNEL_1);
  (void)HAL_TIM_PWM_Start(&htim9, TIM_CHANNEL_2);
  ROBOT_led_set(0U, 0U, 0U);

  /* OLED boot splash (bounded; automatically skipped if OLED is absent) -- */
  OLED_MJ_Init();
  OLED_MJ_mirror_horizental();
  OLED_MJ_mirror_vertical();
  if (ssd1306_GetError() == 0U)
  {
    ssd1306_Fill(Black);
    ssd1306_DrawBitmap(42, 0, fire_bitmap, 64, 40, White);
    OLED_MJ_write_in_middle("Charara", 45, Font_11x18, White);
    ssd1306_UpdateScreen();
    HAL_Delay(400U);
    ssd1306_ClearError();
  }

  /* Watchdog + I2C slave -------------------------------------------------- */
  MX_IWDG_Init();
  ROBOT_led_set(0U, 0U, 32U); /* idle: blue */
  ROBOT_i2c_arm_listen();
  robot_ready = 1U;
}

/** @brief Appends 'value' as decimal text at position 'pos'; bounded, NUL kept. */
static uint8_t ROBOT_fmt_i32(char *buf, uint8_t capacity, uint8_t pos, int32_t value)
{
  char     digits[10];
  uint8_t  count = 0U;
  uint8_t  index;
  uint32_t magnitude;
  uint8_t  negative = 0U;

  if (value < 0)
  {
    negative = 1U;
    magnitude = (uint32_t)(-(value + 1)) + 1U; /* safe for INT32_MIN */
  }
  else
  {
    magnitude = (uint32_t)value;
  }

  do
  {
    digits[count] = (char)('0' + (int)(magnitude % 10U));
    magnitude /= 10U;
    count++;
  } while ((magnitude > 0U) && (count < (uint8_t)sizeof(digits)));

  if (negative != 0U)
  {
    if (pos < capacity)
    {
      buf[pos] = '-';
      pos++;
    }
  }
  for (index = count; index > 0U; index--)
  {
    if (pos < capacity)
    {
      buf[pos] = digits[index - 1U];
      pos++;
    }
  }
  if (pos < capacity)
  {
    buf[pos] = '\0';
  }
  return pos;
}

/** @brief Appends a NUL terminated string into a bounded buffer. */
static uint8_t ROBOT_append_str(char *buf, uint8_t capacity, uint8_t pos, const char *text)
{
  while ((*text != '\0') && (pos < capacity))
  {
    buf[pos] = *text;
    pos++;
    text++;
  }
  if (pos < capacity)
  {
    buf[pos] = '\0';
  }
  return pos;
}

/** @brief Builds the automatic OLED content: state/faults, pose, speed. */
static void ROBOT_build_status_lines(char *line0, char *line1, char *line2)
{
  static const char *const state_names[7] =
  {
    "IDLE", "DRIVE", "TURN", "STRAFE", "GOTO", "FAULT", "ESTOP"
  };
  MOTION_SNAPSHOT snapshot;
  uint8_t pos;

  MOTION_get_snapshot(&snapshot);

  pos = 0U;
  if (snapshot.state < 7U)
  {
    pos = ROBOT_append_str(line0, (uint8_t)ROBOT_OLED_LINE_LEN, pos, state_names[snapshot.state]);
  }
  if (snapshot.faults != 0U)
  {
    pos = ROBOT_append_str(line0, (uint8_t)ROBOT_OLED_LINE_LEN, pos, " F");
    pos = ROBOT_fmt_i32(line0, (uint8_t)ROBOT_OLED_LINE_LEN, pos, (int32_t)snapshot.faults);
  }

  pos = 0U;
  pos = ROBOT_append_str(line1, (uint8_t)ROBOT_OLED_LINE_LEN, pos, "x");
  pos = ROBOT_fmt_i32(line1, (uint8_t)ROBOT_OLED_LINE_LEN, pos, snapshot.x_um / 1000);
  pos = ROBOT_append_str(line1, (uint8_t)ROBOT_OLED_LINE_LEN, pos, " y");
  pos = ROBOT_fmt_i32(line1, (uint8_t)ROBOT_OLED_LINE_LEN, pos, snapshot.y_um / 1000);
  (void)ROBOT_append_str(line1, (uint8_t)ROBOT_OLED_LINE_LEN, pos, "mm");

  pos = 0U;
  pos = ROBOT_append_str(line2, (uint8_t)ROBOT_OLED_LINE_LEN, pos, "th");
  pos = ROBOT_fmt_i32(line2, (uint8_t)ROBOT_OLED_LINE_LEN, pos, snapshot.theta_mdeg / 1000);
  pos = ROBOT_append_str(line2, (uint8_t)ROBOT_OLED_LINE_LEN, pos, " v");
  (void)ROBOT_fmt_i32(line2, (uint8_t)ROBOT_OLED_LINE_LEN, pos, (int32_t)snapshot.v_left_mm_s);
}

/** @brief Throttled OLED refresh: runs only when the bus has been quiet for
 *         ROBOT_I2C_BUS_QUIET_MS, no motion/request is pending; listen mode
 *         is restored on every exit path. */
static void ROBOT_oled_task(uint32_t now_ms)
{
  char  line0[ROBOT_OLED_LINE_LEN];
  char  line1[ROBOT_OLED_LINE_LEN];
  char  line2[ROBOT_OLED_LINE_LEN];
  const char *display0;
  const char *display1;
  const char *display2;

  if (PROTO_get_display_mode() == PROTO_DISPLAY_OFF)
  {
    return;
  }
  if ((int32_t)(now_ms - next_oled_ms) < 0)
  {
    return;
  }
  if (MOTION_is_busy() != 0U)
  {
    next_oled_ms = now_ms + 200U; /* keep the bus free while moving */
    return;
  }
  if (PROTO_has_pending_requests() != 0U)
  {
    return; /* serve the RPi first */
  }
  if (PROTO_ms_since_bus_activity(now_ms) < ROBOT_I2C_BUS_QUIET_MS)
  {
    return;
  }

  if (PROTO_get_display_mode() == PROTO_DISPLAY_AUTO)
  {
    line0[0] = '\0';
    line1[0] = '\0';
    line2[0] = '\0';
    ROBOT_build_status_lines(line0, line1, line2);
    display0 = line0;
    display1 = line1;
    display2 = line2;
  }
  else
  {
    display0 = PROTO_get_display_line(0U);
    display1 = PROTO_get_display_line(1U);
    display2 = PROTO_get_display_line(2U);
  }

  /* Leave slave listen mode for the master transfer to the OLED. */
  (void)HAL_I2C_DisableListen_IT(&hi2c1);
  if (PROTO_bus_is_active() != 0U)
  {
    ROBOT_i2c_arm_listen(); /* the RPi won the race: abort this refresh */
    return;
  }

  ssd1306_Fill(Black);
  ssd1306_SetCursor(2, 0);
  ssd1306_WriteString((char *)display0, Font_7x10, White);
  ssd1306_SetCursor(2, 22);
  ssd1306_WriteString((char *)display1, Font_7x10, White);
  ssd1306_SetCursor(2, 44);
  ssd1306_WriteString((char *)display2, Font_7x10, White);
  ssd1306_UpdateScreen();

  if (ssd1306_GetError() != 0U)
  {
    ssd1306_ClearError();
    next_oled_ms = now_ms + ROBOT_OLED_BACKOFF_MS;
  }
  else
  {
    next_oled_ms = now_ms + ROBOT_OLED_REFRESH_MS;
  }
  ROBOT_i2c_arm_listen();
}

/* --------------------------- Main loop task ------------------------------ */
/** @brief Super-loop tick: protocol, motion, OLED, I2C recovery, LED, IWDG. */
static void ROBOT_task(void)
{
  uint32_t now_ms = HAL_GetTick();
  uint8_t  state;

  PROTO_task();
  MOTION_task();
  ROBOT_oled_task(now_ms);
  ROBOT_i2c_recover();

  /* RGB status: idle blue, moving green, fault red, e-stop blinking. */
  state = MOTION_get_state();
  if ((state == MOTION_STATE_FAULT) || (state == MOTION_STATE_ESTOP))
  {
    if ((state == MOTION_STATE_ESTOP) && (((now_ms / 250U) % 2U) == 0U))
    {
      ROBOT_led_set(0U, 0U, 0U);
    }
    else
    {
      ROBOT_led_set(48U, 0U, 0U);
    }
  }
  else if (MOTION_is_busy() != 0U)
  {
    ROBOT_led_set(0U, 48U, 0U);
  }
  else
  {
    ROBOT_led_set(0U, 0U, 32U);
  }

  (void)HAL_IWDG_Refresh(&hiwdg);
}

/* ======================= I2C listen-mode callbacks ======================= */

/** @brief I2C address matched: arms a 16 B receive (write) or a 32 B
 *         transmit (read) transfer. ISR context. */
void HAL_I2C_AddrCallback(I2C_HandleTypeDef *hi2c, uint8_t TransferDirection, uint16_t AddrMatchCode)
{
  UNUSED(AddrMatchCode);
  if (hi2c->Instance != I2C1)
  {
    return;
  }
  PROTO_note_bus_activity();

  if (TransferDirection == I2C_DIRECTION_TRANSMIT)
  {
    /* Master writes a 16-byte request frame. */
    (void)HAL_I2C_Slave_Seq_Receive_IT(hi2c, PROTO_get_rx_buffer(), PROTO_REQ_SIZE, I2C_LAST_FRAME);
  }
  else
  {
    /* Master reads the 32-byte latest response frame. */
    PROTO_copy_serving_isr();
    (void)HAL_I2C_Slave_Seq_Transmit_IT(hi2c, (uint8_t *)PROTO_get_serving_buffer(), PROTO_RSP_SIZE, I2C_LAST_FRAME);
  }
}

/** @brief Request frame received: protocol validation and queueing. */
void HAL_I2C_SlaveRxCpltCallback(I2C_HandleTypeDef *hi2c)
{
  if (hi2c->Instance != I2C1)
  {
    return;
  }
  PROTO_on_rx_complete_isr(); /* validate, enqueue, immediate safety flags */
}

/** @brief Response frame fully transmitted; nothing further to do. */
void HAL_I2C_SlaveTxCpltCallback(I2C_HandleTypeDef *hi2c)
{
  if (hi2c->Instance != I2C1)
  {
    return;
  }
  /* Nothing to do: the response frame was fully transmitted. */
}

/** @brief Listen mode finished: re-arm it (recovery on failure). */
void HAL_I2C_ListenCpltCallback(I2C_HandleTypeDef *hi2c)
{
  if (hi2c->Instance != I2C1)
  {
    return;
  }
  PROTO_note_bus_activity();
  ROBOT_i2c_arm_listen();
}

/** @brief I2C error: request main-loop recovery (never DeInit in the ISR). */
void HAL_I2C_ErrorCallback(I2C_HandleTypeDef *hi2c)
{
  if (hi2c->Instance != I2C1)
  {
    return;
  }
  /* Never DeInit/Init inside the ISR: flag it and let the main loop recover. */
  i2c_recover_req = 1U;
}

/** @brief EXTI dispatcher: wall limit switches (hard stop) and user buttons. */
void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
  /* Wall limit switches: immediate hard stop + latched fault (ISR context). */
  if ((GPIO_Pin == SW1_Pin) || (GPIO_Pin == SW4_Pin) || (GPIO_Pin == SW3_Pin))
  {
    MOTION_on_limit_switch(GPIO_Pin);
    return;
  }

  if (GPIO_Pin == BUTTON1_Pin)
  {
    button1_counter++;
    if ((button1_counter % 2U) != 0U)
    {
      __HAL_TIM_SET_COMPARE(&htim11, TIM_CHANNEL_1, 255U);
      __HAL_TIM_SET_COMPARE(&htim9, TIM_CHANNEL_1, 255U);
    }
    else
    {
      __HAL_TIM_SET_COMPARE(&htim11, TIM_CHANNEL_1, 0U);
      __HAL_TIM_SET_COMPARE(&htim9, TIM_CHANNEL_1, 0U);
    }
  }
  else if (GPIO_Pin == BUTTON2_Pin)
  {
    button2_counter++;
    if ((button2_counter % 2U) != 0U)
    {
      __HAL_TIM_SET_COMPARE(&htim9, TIM_CHANNEL_1, 255U);
      __HAL_TIM_SET_COMPARE(&htim9, TIM_CHANNEL_2, 255U);
    }
    else
    {
      __HAL_TIM_SET_COMPARE(&htim9, TIM_CHANNEL_1, 0U);
      __HAL_TIM_SET_COMPARE(&htim9, TIM_CHANNEL_2, 0U);
    }
  }
  else
  {
    /* unrelated EXTI line */
  }
}

  * @retval None
  */
void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
  if (htim->Instance == TIM6)
  {
    HAL_IncTick();
    return;
  }
  if (htim->Instance == TIM7)
  {
    /* 100 Hz odometry sampling: refresh the encoder caches first. */
    ENCODERS_MJ_poll();
    MOTION_odometry_tick();
    return;
  }
  /* Side target timers run the stepper driver's open-loop fallback counting
     (normal completion is closed-loop on the odometry encoders). */
  STEPPERS_MJ_set_left_target_callback(htim);
  STEPPERS_MJ_set_right_target_callback(htim);
}

/**
  * @brief  This function is executed in case of error occurrence.
  * @retval None
  */
void Error_Handler(void)
{
  /* Safe state: stop pulses and release the motors if the application is
     already running. The IWDG (started in ROBOT_init) then resets the board
     after ~500 ms; before that the board simply waits for a power cycle. */
  __disable_irq();
  if (robot_ready != 0U)
  {
    STEPPERS_MJ_stop_all();
    STEPPERS_MJ_disable_left_steppers();
    STEPPERS_MJ_disable_right_steppers();
  }
  while (1)
  {
    /* wait for the watchdog reset */
  }
}
